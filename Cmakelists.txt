cmake_minimum_required(VERSION 3.10)

project(Kura)

set(CMAKE_CXX_STANDARD 17)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_SOURCE_DIR}/bin/debug")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin/Release")


add_executable(Kura
    src/kura.c
    src/memory.c
    src/chunk.c
    src/debug.c
    src/vm.c
    src/scanner.c
    src/token.c
    src/compiler.c
    src/value.c
    src/datastruct.c
)

target_compile_definitions(Kura PRIVATE
    $<$<CONFIG:Debug>:DEBUG>
)
